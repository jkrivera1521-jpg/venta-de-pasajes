# Motores de Cálculo Empresariales Avanzados: AviatorScript + Quarkus + PostgreSQL

Este documento contiene tres implementaciones avanzadas diseñadas para arquitecturas de motores de reglas dinámicas basadas en **AviatorScript**, **Quarkus** y **PostgreSQL**. 

Los ejemplos cubren algoritmos de puntuación multidimensional, agregaciones matriciales de ventas jerárquicas y cálculo volumétrico/geográfico logístico, utilizando programación funcional (`seq.map`, `seq.filter`, `seq.reduce`), manejo de matrices anidadas y estructuras de datos complejas.

---

## CASO 1: Motor de Credit Scoring Multidimensional y Evaluación de Capacidad de Pago

### 1. Descripción del Problema
Este motor evalúa solicitudes de crédito de alta complejidad financiera realizando las siguientes operaciones:
1. **Análisis de Capacidad de Pago Dinámica**: Calcula el ratio **DTI (Debt-to-Income)** actual y proyectado, ajustando por variables de estrés macroeconómico e inflación.
2. **Evaluación de Riesgo de Buró de Crédito mediante Funciones de Orden Superior**: Procesa el historial de deudas usando `seq.reduce` y `seq.filter` para identificar moras graves, recurrencia de impagos y endeudamiento total.
3. **Cálculo de LTV (Loan-to-Value) Ajustado por Colateral**: Aplica factores de depreciación según el tipo de garantía prestada.
4. **Matriz de Scoring Multidimensional**: Suma y pondera puntos por estabilidad laboral, comportamiento crediticio, ratio de cobertura y tipo de contrato.
5. **Generación de Plan de Amortización Resumido**: Proyecta la tabla de pagos estimada (amortización francesa) dentro del script ejecutando un ciclo interativo y construyendo una estructura en árbol JSON/Map.

### 3. Script Avanzado en AviatorScript

```javascript
// =========================================================================
// 1. VALIDACIONES DE INTEGRIDAD Y PRECONDICIONES
// =========================================================================
if (solicitudId == nil) { throw "El parámetro 'solicitudId' es obligatorio"; }
if (financiero.ingresoNetoMensual <= 0) { throw "El ingreso neto mensual debe ser mayor que cero"; }
if (propuesta.montoSolicitado <= 0 || propuesta.plazoMeses <= 0) { throw "Monto y plazo deben ser mayores a cero"; }

// =========================================================================
// 2. PROCESAMIENTO DE HISTORIAL CREDITICIO (PROGRAMACIÓN FUNCIONAL)
// =========================================================================
let deudasActivas = seq.filter(historialCrediticio, lambda(x) -> x.estado == "ACTIVO" end);
let deudasCastigadas = seq.filter(historialCrediticio, lambda(x) -> x.estado == "CASTIGADO" || x.diasMoraMaximos > 60 end);

// Verificación de lista negra / castigos directos
if (seq.count(deudasCastigadas) > 0) {
    return seq.map(
        "solicitudId", solicitudId,
        "dictamen", "RECHAZADO_DIRECTO",
        "motivo", "El cliente registra operaciones castigadas o mora superior a 60 días",
        "scoreFinal", 0,
        "montoAprobado", 0.0
    );
}

// Acumuladores de deuda actual usando reduce
let totalCuotasVigentes = seq.reduce(deudasActivas, lambda(acc, item) -> acc + item.cuotaMensual end, 0.0);
let totalDeudaVigente = seq.reduce(deudasActivas, lambda(acc, item) -> acc + item.saldoPendiente end, 0.0);

// =========================================================================
// 3. CÁLCULO DE CUOTA PROYECTADA (FÓRMULA FRANCESA) Y RATIOS
// =========================================================================
let i = propuesta.tasaTNMBase;
let n = propuesta.plazoMeses;
// Factor de amortización: [i * (1+i)^n] / [(1+i)^n - 1]
let factorAmortizacion = (i * math.pow(1 + i, n)) / (math.pow(1 + i, n) - 1);
let cuotaMensualProyectada = propuesta.montoSolicitado * factorAmortizacion;

let ingresoAjustado = financiero.ingresoNetoMensual / financiero.factorEstresInflacionario;
let gastoTotal = financiero.gastosFijosVariables + totalCuotasVigentes + cuotaMensualProyectada;
let dtiProyectado = gastoTotal / ingresoAjustado;

let valorGarantiaReal = propuesta.garantia.valorComercial * propuesta.garantia.factorAjusteLiquidez;
let ltvProyectado = propuesta.montoSolicitado / valorGarantiaReal;

// =========================================================================
// 4. MATRIZ MULTIDIMENSIONAL DE SCORING DE RIESGO
// =========================================================================
let scorePonderado = perfil.scoreBuroInicial;

// Puntos por DTI
if (dtiProyectado <= 0.35) {
    scorePonderado = scorePonderado + 150;
} elsif (dtiProyectado <= 0.50) {
    scorePonderado = scorePonderado + 75;
} elsif (dtiProyectado <= 0.65) {
    scorePonderado = scorePonderado - 50;
} else {
    scorePonderado = scorePonderado - 200;
}

// Puntos por LTV
if (ltvProyectado <= 0.70) {
    scorePonderado = scorePonderado + 100;
} elsif (ltvProyectado <= 0.85) {
    scorePonderado = scorePonderado + 40;
} else {
    scorePonderado = scorePonderado - 100;
}

// Puntos por estabilidad laboral
if (perfil.aniosEstabilidadLaboral >= 5) {
    scorePonderado = scorePonderado + 80;
} elsif (perfil.aniosEstabilidadLaboral >= 2) {
    scorePonderado = scorePonderado + 40;
} else {
    scorePonderado = scorePonderado + 0;
}

// Adjust por tipo de contrato
if (perfil.tipoContrato == "INDEFINIDO") {
    scorePonderado = scorePonderado + 50;
} else {
    scorePonderado = scorePonderado - 30;
}

// =========================================================================
// 5. DETERMINACIÓN DE CONDICIONALES FINALES Y TABLA DE AMORTIZACIÓN
// =========================================================================
let dictamen = "RECHAZADO";
let tasaAjustada = propuesta.tasaTNMBase;
let montoAprobado = 0.0;

if (scorePonderado >= 800 && dtiProyectado <= 0.50) {
    dictamen = "APROBADO_EXPRES";
    tasaAjustada = propuesta.tasaTNMBase * 0.90; // Descuento de tasa por excelente perfil
    montoAprobado = propuesta.montoSolicitado;
} elsif (scorePonderado >= 650 && dtiProyectado <= 0.60) {
    dictamen = "APROBADO_CON_CONDICION";
    tasaAjustada = propuesta.tasaTNMBase * 1.05;
    montoAprobado = propuesta.montoSolicitado * 0.85; // Recorte de monto
} else {
    dictamen = "RECHAZADO_POR_RIESGO";
    montoAprobado = 0.0;
}

// Generación de resumen de cronograma dinámico (primeros 3 meses de simulación)
let cronogramaResumen = seq.list();
if (dictamen != "RECHAZADO_POR_RIESGO") {
    let saldoPendienteIter = montoAprobado;
    let factorAmortAjustado = (tasaAjustada * math.pow(1 + tasaAjustada, n)) / (math.pow(1 + tasaAjustada, n) - 1);
    let cuotaReal = montoAprobado * factorAmortAjustado;

    for periodo in range(1, 4) {
        let interesPeriodo = saldoPendienteIter * tasaAjustada;
        let capitalPeriodo = cuotaReal - interesPeriodo;
        saldoPendienteIter = saldoPendienteIter - capitalPeriodo;

        seq.add(cronogramaResumen, seq.map(
            "periodo", periodo,
            "cuotaTotal", cuotaReal,
            "interes", interesPeriodo,
            "capital", capitalPeriodo,
            "saldoFinal", saldoPendienteIter
        ));
    }
}

// Resultado final estructurado
return seq.map(
    "solicitudId", solicitudId,
    "dictamen", dictamen,
    "scoreCalculado", scorePonderado,
    "ratios", seq.map(
        "dtiProyectado", dtiProyectado,
        "ltvProyectado", ltvProyectado,
        "totalDeudaVigente", totalDeudaVigente
    ),
    "ofertaAprobada", seq.map(
        "montoAprobado", montoAprobado,
        "tasaMensualAplicada", tasaAjustada,
        "plazoMeses", propuesta.plazoMeses,
        "cuotaMensualEstimada", dictamen != "RECHAZADO_POR_RIESGO" ? montoAprobado * ((tasaAjustada * math.pow(1 + tasaAjustada, n)) / (math.pow(1 + tasaAjustada, n) - 1)) : 0.0
    ),
    "resumenCronogramaInicial", cronogramaResumen
);
```

## CASO 2: Motor de Liquidación de Comisiones Corporativas y Bonos Jerárquicos

### 1. Descripción del Problema
Este motor calcula las comisiones y bonificaciones para ejecutivos de venta multinivel considerando:
1. **Consolidación de Ventas Directas e Indirectas**: Procesa la cartera de productos vendidos clasificándolos por categoría (Software, Hardware, Servicios).
2. **Cumplimiento de Meta (KPI Attainment)**: Evalúa el desempeño porcentual respecto a la cuota asignada.
3. **Aceleradores Escalonados Dinámicos**: Aplica porcentajes marginales por tramos (Marginal Tiering Strategy).
4. **Bono por Sobre-Cumplimiento Jerárquico**: Calcula un incentivo sobre el equipo a cargo si se lidera una unidad de negocio.
5. **Deducción por Cancelaciones (Clawbacks)**: Aplica penalizaciones financieras por ventas revertidas o impagadas.

### 2. Parámetros de Entrada (JSON)

```json
{
  "ejecutivo": {
    "id": "EXEC-0092",
    "rol": "DIRECTOR_COMERCIAL",
    "metaCuotaGlobal": 100000.00,
    "porcentajeComisionEquipo": 0.025
  },
  "ventasDirectas": [
    {"idVenta": "V-101", "categoria": "SOFTWARE", "monto": 45000.00, "moraCobro": false},
    {"idVenta": "V-102", "categoria": "SERVICIOS", "monto": 25000.00, "moraCobro": false},
    {"idVenta": "V-103", "categoria": "HARDWARE", "monto": 50000.00, "moraCobro": false},
    {"idVenta": "V-104", "categoria": "SOFTWARE", "monto": 15000.00, "moraCobro": true}
  ],
  "ventasEquipoSubordinado": [
    {"subordinadoId": "SUB-01", "montoTotal": 80000.00, "metaCumplida": true},
    {"subordinadoId": "SUB-02", "montoTotal": 60000.00, "metaCumplida": false}
  ],
  "clawbacks": [
    {"idVentaRevertida": "V-089", "montoComisionAbonadaPreviamente": 1200.00, "motivo": "CANCELACION_CONTRATO"}
  ]
}
```

### 3. Script Avanzado en AviatorScript

```javascript
// =========================================================================
// 1. FILTRADO Y PROCESAMIENTO DE VENTAS DIRECTAS
// =========================================================================
let ventasValidas = seq.filter(ventasDirectas, lambda(v) -> v.moraCobro == false end);
let ventasEnMora = seq.filter(ventasDirectas, lambda(v) -> v.moraCobro == true end);

// Sumatoria total de ventas directas válidas
let totalVendidoDirecto = seq.reduce(ventasValidas, lambda(acc, v) -> acc + v.monto end, 0.0);

// Agregación de ventas por categoría usando seq.reduce
let totalSoftware = seq.reduce(
    seq.filter(ventasValidas, lambda(v) -> v.categoria == "SOFTWARE" end),
    lambda(acc, v) -> acc + v.monto end, 0.0
);
let totalServicios = seq.reduce(
    seq.filter(ventasValidas, lambda(v) -> v.categoria == "SERVICIOS" end),
    lambda(acc, v) -> acc + v.monto end, 0.0
);
let totalHardware = seq.reduce(
    seq.filter(ventasValidas, lambda(v) -> v.categoria == "HARDWARE" end),
    lambda(acc, v) -> acc + v.monto end, 0.0
);

// =========================================================================
// 2. APLICACIÓN DE ACELERADORES POR TRAMOS MARGINALES
// =========================================================================
let tasaCumplimiento = totalVendidoDirecto / ejecutivo.metaCuotaGlobal;
let comisionBaseDirecta = 0.0;

// Tramos marginales de comisión
let montoRestante = totalVendidoDirecto;

// Tramo 1: Hasta el 80% de la meta (Tasa 3%)
let tramo1Cap = ejecutivo.metaCuotaGlobal * 0.80;
if (montoRestante > 0) {
    let montoTramo1 = math.min(montoRestante, tramo1Cap);
    comisionBaseDirecta = comisionBaseDirecta + (montoTramo1 * 0.03);
    montoRestante = montoRestante - montoTramo1;
}

// Tramo 2: Del 80% al 100% de la meta (Tasa 6%)
let tramo2Cap = ejecutivo.metaCuotaGlobal * 0.20;
if (montoRestante > 0) {
    let montoTramo2 = math.min(montoRestante, tramo2Cap);
    comisionBaseDirecta = comisionBaseDirecta + (montoTramo2 * 0.06);
    montoRestante = montoRestante - montoTramo2;
}

// Tramo 3: Excedente > 100% (Acelerador Premium Tasa 10%)
if (montoRestante > 0) {
    comisionBaseDirecta = comisionBaseDirecta + (montoRestante * 0.10);
}

// Factor multiplicador por mix de software (Incentivo de margen alto)
let ratioSoftware = totalSoftware / (totalVendidoDirecto == 0 ? 1.0 : totalVendidoDirecto);
let bonoBonusMixSoftware = 0.0;
if (ratioSoftware >= 0.40) {
    bonoBonusMixSoftware = comisionBaseDirecta * 0.15; // 15% adicional sobre la comisión
}

// =========================================================================
// 3. COMISIÓN POR EQUIPO Y LIDERAZGO
// =========================================================================
let bonoEquipo = 0.0;
if (ejecutivo.rol == "DIRECTOR_COMERCIAL" || ejecutivo.rol == "GERENTE_VENTAS") {
    let totalVendidoEquipo = seq.reduce(ventasEquipoSubordinado, lambda(acc, sub) -> acc + sub.montoTotal end, 0.0);
    bonoEquipo = totalVendidoEquipo * ejecutivo.porcentajeComisionEquipo;
}

// =========================================================================
// 4. CÁLCULO DE CLAWBACKS Y PENALIZACIONES
// =========================================================================
let totalClawback = seq.reduce(clawbacks, lambda(acc, c) -> acc + c.montoComisionAbonadaPreviamente end, 0.0);

// Deducción por ventas en mora sin cobro
let penalizacionMora = seq.reduce(ventasEnMora, lambda(acc, v) -> acc + (v.monto * 0.02) end, 0.0);

// =========================================================================
// 5. LIQUIDACIÓN NETO FINAL Y ESTRUCTURA DE RESPUESTA
// =========================================================================
let totalIngresosBrutosComision = comisionBaseDirecta + bonoBonusMixSoftware + bonoEquipo;
let totalDeducciones = totalClawback + penalizacionMora;
let liquidacionNeto = totalIngresosBrutosComision - totalDeducciones;

return seq.map(
    "ejecutivoId", ejecutivo.id,
    "desempeno", seq.map(
        "totalVendidoDirecto", totalVendidoDirecto,
        "porcentajeCumplimientoMeta", tasaCumplimiento * 100.0,
        "ratioVentaSoftware", ratioSoftware
    ),
    "desgloseComisiones", seq.map(
        "comisionDirectaEscalonada", comisionBaseDirecta,
        "bonoMixSoftware", bonoBonusMixSoftware,
        "bonoLiderazgoEquipo", bonoEquipo,
        "totalBruto", totalIngresosBrutosComision
    ),
    "deducciones", seq.map(
        "clawbacksVentasCanceladas", totalClawback,
        "penalizacionVentasMora", penalizacionMora,
        "totalDeducciones", totalDeducciones
    ),
    "netoAPagar", liquidacionNeto < 0 ? 0.0 : liquidacionNeto,
    "saldoPendienteRetenido", liquidacionNeto < 0 ? math.abs(liquidacionNeto) : 0.0
);
```

## CASO 3: Tarifario Logístico B2B Volumétrico y Tarificación Geográfica Dinámica

### 1. Descripción del Problema
Este motor realiza la cotización de envíos de carga pesada y logística multimodal mediante las siguientes reglas:
1. **Cálculo de Peso Volumétrico Dimensional**: Evalúa la relación entre peso real y peso por volumen $(Alto \times Ancho \times Largo) / FactorDim$.
2. **Cálculo de Matriz Distancia/Zonificación**: Aplica una tarifa base según origen y destino, añadiendo peajes interprovinciales.
3. **Cálculo de Recargos por Carga Especial**: Aplica coeficientes multiplicadores dinámicos por mercancía peligrosa (HazMat), refrigeración o fragilidad.
4. **Descuento por Escalado de Volumen Frecuente**: Deducción progresiva si la orden incluye un consolidado de paquetes múltiples (`paquetesList`).
5. **Impuestos Regionales Dinámicos**: Aplica tasas diferenciadas de IVA e impuestos locales según la zona geográfica de entrega.

### 3. Script Avanzado en AviatorScript

```javascript
// =========================================================================
// 1. VALIDACIÓN Y PROCESAMIENTO PAQUETE POR PAQUETE (ITERACIÓN)
// =========================================================================
if (seq.count(paquetes) == 0) { throw "La orden logícstica debe incluir al menos un paquete"; }

let factorDim = cliente.factorVolumetricoSector <= 0 ? 5000.0 : cliente.factorVolumetricoSector;
let detalleCalculoPaquetes = seq.list();

let totalPesoFacturableGlobal = 0.0;
let totalRecargosEspecialesPaquetes = 0.0;

for pkg in paquetes {
    let volumenCc = pkg.altoCm * pkg.anchoCm * pkg.largoCm;
    let pesoVolumetrico = volumenCc / factorDim;
    
    // El peso facturable es el mayor entre peso real y peso volumétrico
    let pesoFacturablePkg = math.max(pkg.pesoRealKg, pesoVolumetrico);
    totalPesoFacturableGlobal = totalPesoFacturableGlobal + pesoFacturablePkg;

    // Recargos individuales
    let recargoHazMat = pkg.esHazMat ? 45.00 : 0.0;
    let recargoRefrigerado = pkg.requiereRefrigeracion ? 30.00 : 0.0;
    
    let subtotalRecargosPkg = recargoHazMat + recargoRefrigerado;
    totalRecargosEspecialesPaquetes = totalRecargosEspecialesPaquetes + subtotalRecargosPkg;

    seq.add(detalleCalculoPaquetes, seq.map(
        "pesoReal", pkg.pesoRealKg,
        "pesoVolumetrico", pesoVolumetrico,
        "pesoFacturableAplicado", pesoFacturablePkg,
        "recargosEspeciales", subtotalRecargosPkg
    ));
}

// =========================================================================
// 2. MATRIZ DE TARIFICACIÓN POR ZONA Y DISTANCIA
// =========================================================================
let tarifaBasePorKm = 0.0;

if (origen.zonaId == "ZONA_NORTE" && destino.zonaId == "ZONA_SUR_RURAL") {
    tarifaBasePorKm = 1.25;
} elsif (origen.zonaId == destino.zonaId) {
    tarifaBasePorKm = 0.75;
} else {
    tarifaBasePorKm = 1.00;
}

let costoFleteBase = destino.distanciaKm * tarifaBasePorKm;
let costoPesoBruto = totalPesoFacturableGlobal * 0.85;

// =========================================================================
// 3. RECARGOS GEOGRÁFICOS Y DE ENTORNO
// =========================================================================
let recargoRuralidad = destino.esUrbano == false ? 25.00 : 0.0;
let recargoCombustible = (costoFleteBase + costoPesoBruto) * condicionesEntorno.recargoCombustiblePorcentaje;
let recargoTemporada = condicionesEntorno.temporadaAlta ? (costoFleteBase * 0.12) : 0.0;

// =========================================================================
// 4. DESCUENTO CORPORATIVO Y TIER B2B
// =========================================================================
let subtotalSinDescuento = costoFleteBase + costoPesoBruto + totalRecargosEspecialesPaquetes + recargoRuralidad + recargoCombustible + recargoTemporada;

let porcentajeDescuentoCliente = 0.0;
if (cliente.tipoCliente == "VIP_B2B") {
    porcentajeDescuentoCliente = 0.15;
} elsif (cliente.tipoCliente == "REGULAR_B2B") {
    porcentajeDescuentoCliente = 0.05;
}

let montoDescuento = subtotalSinDescuento * porcentajeDescuentoCliente;
let subtotalConDescuento = subtotalSinDescuento - montoDescuento;

// =========================================================================
// 5. IMPUESTOS REGIONALES Y ESTRUCTURA DE SALIDA
// =========================================================================
let tasaIva = 0.12;
let montoIva = subtotalConDescuento * tasaIva;
let totalCotizacionFinal = subtotalConDescuento + montoIva;

return seq.map(
    "envioId", envioId,
    "resumenPesos", seq.map(
        "totalPaquetes", seq.count(paquetes),
        "pesoFacturableTotalKg", totalPesoFacturableGlobal
    ),
    "desgloseTarifario", seq.map(
        "fleteBaseKm", costoFleteBase,
        "costoPorPesoTotal", costoPesoBruto,
        "totalRecargosMercancia", totalRecargosEspecialesPaquetes,
        "recargoRuralidad", recargoRuralidad,
        "recargoCombustible", recargoCombustible,
        "recargoTemporadaAlta", recargoTemporada
    ),
    "mecanismoDescuento", seq.map(
        "tipoCliente", cliente.tipoCliente,
        "montoDescuentoAplicado", montoDescuento
    ),
    "totales", seq.map(
        "subtotalNeto", subtotalConDescuento,
        "iva", montoIva,
        "totalPagar", totalCotizacionFinal
    ),
    "detalleIndividualPaquetes", detalleCalculoPaquetes
);
```
