# Motores de Cálculo Nivel Arquitecto/Experto: Turismo y Transporte

Este documento presenta implementaciones de **alta complejidad** en **AviatorScript** orientadas a la **industria turística (Agencias de Viajes / OTAs)** y **empresas de transporte de pasajeros y carga**, listas para integrarse en **Quarkus** con persistencia en **PostgreSQL**.

## CASO 1: Motor de Tarificación Dinámica, Dynamic Packaging y Yield Management para Agencia de Viajes (OTA)

### 1. Descripción del Problema
Este motor cotiza paquetes turísticos a la medida (Vuelo + Hotel + Excursiones + Seguro) aplicando:
1. **Normalización Cambiaria y Márgenes por Proveedor**: Convierte automáticamente costos en múltiples monedas (USD, EUR, BRL) con tasas fijadas y aplica márgenes diferenciados (`seq.map`, `seq.reduce`).
2. **Curva de Tarificación Dinámica (Yield Management)**: Aplica un multiplicador de anticipación (Lead Time) y ocupación de temporada alta/baja.
3. **Descuentos por Composición Familiar y Pasajeros**: Categorización funcional de pasajeros (Adulto, Niño, Infant) con reglas de gratuidad y tasas fijas.
4. **Matriz de Políticas de Cancelación y Penalización**: Genera una matriz proyectada de penalidades según la fecha límite de cancelación.

### 2. Estructura de Parámetros de Entrada (JSON)

```json
{
  "cotizacionId": "COT-TRAVEL-2026-8801",
  "diasHastaViaje": 12,
  "temporada": "ALTA",
  "tasasCambioUSD": {
    "USD": 1.0,
    "EUR": 1.08,
    "BRL": 0.18
  },
  "pasajeros": [
    {"id": "PAX-01", "tipo": "ADULTO", "edad": 35},
    {"id": "PAX-02", "tipo": "ADULTO", "edad": 33},
    {"id": "PAX-03", "tipo": "NINO",   "edad": 7},
    {"id": "PAX-04", "tipo": "INFANT", "edad": 1}
  ],
  "servicios": [
    {"tipo": "VUELO", "proveedor": "LATAM", "costoNetoUnitario": 350.0, "moneda": "USD", "markupEsperado": 0.12},
    {"tipo": "HOTEL", "proveedor": "MARRIOTT", "costoNetoUnitario": 120.0, "moneda": "EUR", "markupEsperado": 0.18, "noches": 5},
    {"tipo": "TOUR",  "proveedor": "CIVITATIS", "costoNetoUnitario": 250.0, "moneda": "BRL", "markupEsperado": 0.25}
  ],
  "parametroComisionAgencia": 0.08
}
```

### 3. Script Avanzado en AviatorScript

```javascript
// =========================================================================
// 1. VALIDACIONES INICIALES Y CONTEO DE PASAJEROS
// =========================================================================
if (cotizacionId == nil) { throw "Identificador de cotización no proporcionado"; }
if (seq.count(pasajeros) == 0) { throw "La lista de pasajeros no puede estar vacía"; }
if (seq.count(servicios) == 0) { throw "Debe incluir al menos un servicio en el paquete"; }

let numAdultos = seq.count(seq.filter(pasajeros, lambda(p) -> p.tipo == "ADULTO" end));
let numNinos   = seq.count(seq.filter(pasajeros, lambda(p) -> p.tipo == "NINO" end));
let numInfants = seq.count(seq.filter(pasajeros, lambda(p) -> p.tipo == "INFANT" end));

if (numAdultos == 0) { throw "Todo paquete turístico debe incluir al menos un adulto responsable"; }

// =========================================================================
// 2. FACTOR DE YIELD MANAGEMENT (ANTICIPACIÓN Y TEMPORADA)
// =========================================================================
let factorAnticipacion = 1.0;
if (diasHastaViaje < 7) {
    factorAnticipacion = 1.25; // Compra de último minuto
} elsif (diasHastaViaje < 30) {
    factorAnticipacion = 1.10;
} elsif (diasHastaViaje > 90) {
    factorAnticipacion = 0.92; // Descuento por reserva temprana (Early Bird)
}

let factorTemporada = temporada == "ALTA" ? 1.30 : (temporada == "MEDIA" ? 1.10 : 0.85);
let multiplicadorYieldGlobal = factorAnticipacion * factorTemporada;

// =========================================================================
// 3. LIQUIDACIÓN DE SERVICIOS Y CONVERSIÓN MULTIMONEDA
// =========================================================================
let detalleServiciosCotizados = seq.list();

for srv in servicios {
    let tasaConversion = tasasCambioUSD[srv.moneda];
    if (tasaConversion == nil || tasaConversion <= 0) {
        throw "Tasa de cambio no disponible para la moneda: " + srv.moneda;
    }

    let costoNetoUSD = srv.costoNetoUnitario * tasaConversion;
    
    // Multiplicador por noches si aplica
    let factorCantidad = srv.tipo == "HOTEL" ? srv.noches : 1.0;
    let costoBaseServicioTotal = costoNetoUSD * factorCantidad;

    // Acomodación de tarifa por tipo de pasajero (Adulto 100%, Niño 65%, Infant 10%)
    let costoEfectivoPasajeros = (numAdultos * costoBaseServicioTotal) 
                               + (numNinos * costoBaseServicioTotal * 0.65) 
                               + (numInfants * costoBaseServicioTotal * 0.10);

    let precioVentaServicio = (costoEfectivoPasajeros * (1.0 + srv.markupEsperado)) * multiplicadorYieldGlobal;

    seq.add(detalleServiciosCotizados, seq.map(
        "tipoServicio", srv.tipo,
        "proveedor", srv.proveedor,
        "monedaOrigen", srv.moneda,
        "costoNetoUSD", costoEfectivoPasajeros,
        "precioVentaFinalUSD", precioVentaServicio,
        "margenGananciaBrutaUSD", precioVentaServicio - costoEfectivoPasajeros
    ));
}

// =========================================================================
// 4. AGREGACIÓN DE TOTALES Y APLICACIÓN DE COMISIÓN DE AGENCIA
// =========================================================================
let costoNetoTotalPaqueteUSD = seq.reduce(detalleServiciosCotizados, lambda(acc, item) -> acc + item.costoNetoUSD end, 0.0);
let precioVentaBasePaqueteUSD = seq.reduce(detalleServiciosCotizados, lambda(acc, item) -> acc + item.precioVentaFinalUSD end, 0.0);

let valorComisionAgenciaUSD = precioVentaBasePaqueteUSD * parametroComisionAgencia;
let precioVentaTotalAgenciaUSD = precioVentaBasePaqueteUSD + valorComisionAgenciaUSD;
let impuestosTurismoLocales = precioVentaTotalAgenciaUSD * 0.05; // 5% Tasa de Turismo

let granTotalPagarUSD = precioVentaTotalAgenciaUSD + impuestosTurismoLocales;
let totalPasajerosPago = numAdultos + numNinos; // Infants no ocupan asiento/cama completa
let costoPromedioPorPasajeroUSD = granTotalPagarUSD / totalPasajerosPago;

// =========================================================================
// 5. MATRIZ DE POLÍTICAS DE CANCELACIÓN PROYECTADAS
// =========================================================================
let tablaPenalizacionesCancelacion = seq.list();

seq.add(tablaPenalizacionesCancelacion, seq.map(
    "tramoDias", "30+ días antes",
    "porcentajePenalizacion", 0.10,
    "montoPenalidadUSD", granTotalPagarUSD * 0.10,
    "montoReembolsableUSD", granTotalPagarUSD * 0.90
));

seq.add(tablaPenalizacionesCancelacion, seq.map(
    "tramoDias", "15 a 29 días antes",
    "porcentajePenalizacion", 0.35,
    "montoPenalidadUSD", granTotalPagarUSD * 0.35,
    "montoReembolsableUSD", granTotalPagarUSD * 0.65
));

seq.add(tablaPenalizacionesCancelacion, seq.map(
    "tramoDias", "0 a 14 días antes",
    "porcentajePenalizacion", 1.00,
    "montoPenalidadUSD", granTotalPagarUSD,
    "montoReembolsableUSD", 0.0
));

// =========================================================================
// 6. ESTRUCTURA DE RETORNO COMPLEJA
// =========================================================================
return seq.map(
    "cotizacionId", cotizacionId,
    "resumenPasajeros", seq.map(
        "adultos", numAdultos,
        "ninos", numNinos,
        "infants", numInfants,
        "totalFacturable", totalPasajerosPago
    ),
    "factoresYieldAplicados", seq.map(
        "factorAnticipacion", factorAnticipacion,
        "factorTemporada", factorTemporada,
        "multiplicadorYieldGlobal", multiplicadorYieldGlobal
    ),
    "desgloseServicios", detalleServiciosCotizados,
    "resumenFinanciero", seq.map(
        "costoNetoProveedorTotalUSD", costoNetoTotalPaqueteUSD,
        "comisionAgenciaUSD", valorComisionAgenciaUSD,
        "impuestosTurismoUSD", impuestosTurismoLocales,
        "granTotalCotizacionUSD", granTotalPagarUSD,
        "precioPromedioPorPersonaUSD", costoPromedioPorPasajeroUSD
    ),
    "politicaCancelacionProyectada", tablaPenalizacionesCancelacion
);
```

### 4. Persistencia SQL (PostgreSQL)

```sql
INSERT INTO calculos.formula (
    codigo,
    nombre,
    version,
    expresion,
    activa,
    fecha_creacion
) VALUES (
    'FORM-TURISMO-DYNAMIC-PACKAGING-V1',
    'Motor Dynamic Packaging y Yield Management para Agencias de Viaje',
    1,
    $$
    if (cotizacionId == nil) { throw "Cotización requerida"; }
    let numAdultos = seq.count(seq.filter(pasajeros, lambda(p) -> p.tipo == "ADULTO" end));
    let numNinos   = seq.count(seq.filter(pasajeros, lambda(p) -> p.tipo == "NINO" end));
    
    let factorAnticipacion = diasHastaViaje < 7 ? 1.25 : 1.0;
    let factorTemporada = temporada == "ALTA" ? 1.30 : 1.0;
    let yieldFactor = factorAnticipacion * factorTemporada;

    let serviciosCalculados = seq.map(servicios, lambda(srv) ->
        let tasa = tasasCambioUSD[srv.moneda];
        let costoNeto = srv.costoNetoUnitario * tasa * (numAdultos + (numNinos * 0.65));
        let precioVenta = costoNeto * (1.0 + srv.markupEsperado) * yieldFactor;
        return seq.map("tipo", srv.tipo, "costoUSD", costoNeto, "precioUSD", precioVenta);
    end);

    let totalNeto = seq.reduce(serviciosCalculados, lambda(acc, s) -> acc + s.costoUSD end, 0.0);
    let totalVenta = seq.reduce(serviciosCalculados, lambda(acc, s) -> acc + s.precioUSD end, 0.0);

    return seq.map(
        "cotizacionId", cotizacionId,
        "totalNetoUSD", totalNeto,
        "totalVentaUSD", totalVenta,
        "margenUSD", totalVenta - totalNeto,
        "servicios", serviciosCalculados
    );
    $$,
    TRUE,
    CURRENT_TIMESTAMP
);
```

## CASO 2: Motor de Tarificación de Boletos Interprovinciales, Carga Encomiendas y Sobrestante de Flota

### 1. Descripción del Problema
Este motor realiza la cotización completa para una unidad de transporte interprovincial de pasajeros y carga combinada:
1. **Dynamic Seat Pricing por Factor de Ocupación**: Calcula el costo del billete de pasajero en función del porcentaje de asientos vendidos al momento de la consulta.
2. **Tarificación Volumétrica y Peso Carga/Equipaje Excedente**: Aplica la regla del Peso Volumétrico ($\frac{\text{Largo} \times \text{Ancho} \times \text{Alto}}{5000}$) para equipaje o encomiendas.
3. **Recargo Operativo por Combustible y Peajes de Ruta**: Distribuye proporcionalmente costos de peaje y recargo de combustible diésel según la distancia en kilómetros.
4. **Descuento por Ley de Discapacidad / Tercera Edad**: Aplica exenciones legales del 50% de tarifa sobre la base imponible.

### 2. Estructura de Parámetros de Entrada (JSON)

```json
{
  "viajeId": "VIAJE-EXPRESO-2026-402",
  "ruta": {
    "origen": "QUITO",
    "destino": "GUAYAQUIL",
    "distanciaKm": 420.0,
    "numeroPeajes": 5,
    "costoPromedioPeaje": 1.50
  },
  "capacidadBus": {
    "totalAsientos": 40,
    "asientosOcupadosActuales": 32
  },
  "boletoSolicitado": {
    "categoriaAsiento": "VIP_PISO_1",
    "tipoPasajero": "TERCERA_EDAD",
    "tarifaBaseRuta": 18.00
  },
  "encomiendasYEquipaje": [
    {"descripcion": "Maleta Viaje", "pesoKg": 18.0, "largoCm": 60, "anchoCm": 40, "altoCm": 30, "esEquipajeMano": true},
    {"descripcion": "Caja Repuestos", "pesoKg": 32.0, "largoCm": 80, "anchoCm": 60, "altoCm": 50, "esEquipajeMano": false}
  ],
  "combustibleParametros": {
    "precioGalonDiesel": 1.90,
    "rendimientoKmPorGalon": 12.0
  }
}
```

### 3. Script Avanzado en AviatorScript

```javascript
// =========================================================================
// 1. VALIDACIÓN DE ASORTMENT Y DISPONIBILIDAD
// =========================================================================
if (viajeId == nil) { throw "Identificador de viaje obligatorio"; }
if (capacidadBus.asientosOcupadosActuales >= capacidadBus.totalAsientos) {
    throw "El autobús se encuentra al 100% de su capacidad. No hay asientos disponibles.";
}

// =========================================================================
// 2. CÁLCULO DIVERSIFICADO DE TARIFA DE PASAJERO (DYNAMIC SEAT PRICING)
// =========================================================================
let porcentajeOcupacion = (capacidadBus.asientosOcupadosActuales * 1.0) / capacidadBus.totalAsientos;

let factorDynamicPricing = 1.0;
if (porcentajeOcupacion >= 0.90) {
    factorDynamicPricing = 1.40; // Últimos asientos disponibles
} elsif (porcentajeOcupacion >= 0.70) {
    factorDynamicPricing = 1.20;
} elsif (porcentajeOcupacion < 0.30) {
    factorDynamicPricing = 0.90; // Descuento de incentivo de llenado
}

let recargoCategoriaAsiento = boletoSolicitado.categoriaAsiento == "VIP_PISO_1" ? 1.25 : 1.0;
let tarifaSubtotalAsiento = boletoSolicitado.tarifaBaseRuta * factorDynamicPricing * recargoCategoriaAsiento;

// Aplicación de Descuento Legal (Tercera Edad / Discapacidad / Niños)
let porcentajeDescuentoLegal = 0.0;
if (boletoSolicitado.tipoPasajero == "TERCERA_EDAD" || boletoSolicitado.tipoPasajero == "DISCAPACIDAD") {
    porcentajeDescuentoLegal = 0.50;
} elsif (boletoSolicitado.tipoPasajero == "NINO") {
    porcentajeDescuentoLegal = 0.30;
}

let valorDescuentoLegal = tarifaSubtotalAsiento * porcentajeDescuentoLegal;
let tarifaFinalBoletoPasajero = tarifaSubtotalAsiento - valorDescuentoLegal;

// =========================================================================
// 3. TARIFICACIÓN DE ENCOMIENDAS Y EXCESO DE EQUIPAJE (VOLUMÉTRICO)
// =========================================================================
let detalleCargaCalculada = seq.list();
let totalCargaEquipajeUSD = 0.0;

for item in encomiendasYEquipaje {
    // Franquicia de Equipaje: Equipaje de mano o hasta 15kg es GRATIS
    let esGratis = item.esEquipajeMano || item.pesoKg <= 15.0;
    
    let pesoVolumetrico = (item.largoCm * item.anchoCm * item.altoCm) / 5000.0;
    let pesoFacturable = math.max(item.pesoKg * 1.0, pesoVolumetrico);
    
    let costoItemCarga = 0.0;
    if (!esGratis) {
        let pesoExcedente = math.max(0.0, pesoFacturable - 15.0);
        costoItemCarga = (pesoExcedente * 0.75) + (ruta.distanciaKm * 0.01);
    }
    
    totalCargaEquipajeUSD = totalCargaEquipajeUSD + costoItemCarga;

    seq.add(detalleCargaCalculada, seq.map(
        "descripcion", item.descripcion,
        "pesoRealKg", item.pesoKg,
        "pesoVolumetricoKg", pesoVolumetrico,
        "pesoFacturableKg", pesoFacturable,
        "exentoPago", esGratis,
        "costoFleteUSD", costoItemCarga
    ));
}

// =========================================================================
// 4. RECARGOS OPERATIVOS (COMBUSTIBLE Y PEAJES PROPORCIONALES)
// =========================================================================
let galonesEstimadosRuta = ruta.distanciaKm / combustibleParametros.rendimientoKmPorGalon;
let costoTotalCombustibleRuta = galonesEstimadosRuta * combustibleParametros.precioGalonDiesel;
let costoTotalPeajesRuta = ruta.numeroPeajes * ruta.costoPromedioPeaje;

// Distribución proporcional del costo operativo por pasajero
let recargoOperativoPorPasajero = (costoTotalCombustibleRuta + costoTotalPeajesRuta) / capacidadBus.totalAsientos;

// =========================================================================
// 5. CONSOLIDACIÓN DE TOTALES Y RETORNO
// =========================================================================
let totalPagarPasajeYCarga = tarifaFinalBoletoPasajero + totalCargaEquipajeUSD + recargoOperativoPorPasajero;
let seguroPasajeroObligatorio = 1.25;
let granTotalTicketUSD = totalPagarPasajeYCarga + seguroPasajeroObligatorio;

return seq.map(
    "viajeId", viajeId,
    "resumenRuta", seq.map(
        "origen", ruta.origen,
        "destino", ruta.destino,
        "distanciaKm", ruta.distanciaKm,
        "ocupacionBusPorcentaje", porcentajeOcupacion * 100.0
    ),
    "liquidacionBoleto", seq.map(
        "tarifaBase", boletoSolicitado.tarifaBaseRuta,
        "factorOcupacionAplicado", factorDynamicPricing,
        "subtotalSinDescuento", tarifaSubtotalAsiento,
        "descuentoLegalAplicado", valorDescuentoLegal,
        "tarifaNetaPasaje", tarifaFinalBoletoPasajero
    ),
    "desgloseCargaYEncomiendas", detalleCargaCalculada,
    "cargosEspecialesYOperativos", seq.map(
        "totalFleteCargaUSD", totalCargaEquipajeUSD,
        "cuotaOperativaCombustiblePeajeUSD", recargoOperativoPorPasajero,
        "seguroObligatorioPasajeroUSD", seguroPasajeroObligatorio
    ),
    "montoTotalTicketUSD", granTotalTicketUSD
);
```

### 4. Persistencia SQL (PostgreSQL)

```sql
INSERT INTO calculos.formula (
    codigo,
    nombre,
    version,
    expresion,
    activa,
    fecha_creacion
) VALUES (
    'FORM-TRANSPORTE-TARIFICACION-V1',
    'Motor de Tarificación Dinámica de Boletos y Cargo Volumétrico',
    1,
    $$
    if (viajeId == nil) { throw "Viaje no especificado"; }
    
    let ocupacion = (capacidadBus.asientosOcupadosActuales * 1.0) / capacidadBus.totalAsientos;
    let factorDynamic = ocupacion >= 0.85 ? 1.30 : 1.0;
    
    let tarifaBase = boletoSolicitado.tarifaBaseRuta * factorDynamic;
    let descLegal = (boletoSolicitado.tipoPasajero == "TERCERA_EDAD") ? tarifaBase * 0.50 : 0.0;
    let tarifaPasaje = tarifaBase - descLegal;

    let fleteCarga = seq.reduce(encomiendasYEquipaje, lambda(acc, item) ->
        let pVol = (item.largoCm * item.anchoCm * item.altoCm) / 5000.0;
        let pFact = math.max(item.pesoKg * 1.0, pVol);
        return acc + (pFact > 15.0 ? (pFact - 15.0) * 0.75 : 0.0);
    end, 0.0);

    return seq.map(
        "viajeId", viajeId,
        "factorOcupacion", ocupacion,
        "tarifaPasajeUSD", tarifaPasaje,
        "fleteCargaUSD", fleteCarga,
        "totalPagarUSD", tarifaPasaje + fleteCarga + 1.25
    );
    $$,
    TRUE,
    CURRENT_TIMESTAMP
);
```
