# Motores de Cálculo Nivel Arquitecto/Experto: AviatorScript + Quarkus + PostgreSQL

Este documento presenta implementaciones de **complejidad avanzada** diseñadas para explotar la máxima capacidad funcional de **AviatorScript** integrado en un microservicio **Quarkus** con persistencia en **PostgreSQL**.

Los casos abordan algoritmos de **optimización actuarial/financiera**, **motores de precios dinámicos para energía/utilities** y **cálculos de reaseguro y siniestralidad matricial**, utilizando:
- Lambdas complejas anidadas (`seq.reduce`, `seq.filter`, `seq.map`, `seq.sort`).
- Algoritmos numéricos iterativos (metodologías de amortización con tasa interna ajustada y optimización en ciclos).
- Estructuras de datos dinámicas profundas (mapas ordenados, acumulación de listas matriciales).
- Validaciones defensivas y control de excepciones (`throw`, `nil` checks, predicados defensivos).

## CASO 1: Motor Actuarial de Evaluación de Riesgo de Seguros de Vida y Salud Colectivos

### 1. Descripción del Problema
Este motor realiza la cotización actuarial para pólizas corporativas masivas evaluando:
1. **Análisis Estadístico Poblacional**: Procesa la nomina de asegurados calculando varianza de edad, mediana y factores de mortalidad ajustados mediante funciones lambda.
2. **Factor de Morbilidad por Comorbilidades**: Aplica filtros funcionales (`seq.filter`) sobre listas de diagnósticos previos y calcula un índice numérico de riesgo ponderado por edad.
3. **Ajuste de Carga de Siniestralidad Histórica**: Calcula el ratio de pérdida histórica (Loss Ratio) mediante reducción acumulativa (`seq.reduce`).
4. **Cálculo de Prima Pura de Riesgo con Curva Exponencial**: Implementa funciones de la librería `math` (`math.pow`, `math.exp`) para simular la aceleración de riesgo por encima de los 50 años.
5. **Simulación Montecarlo / Tabla de Proyección de Reservas**: Genera una matriz dinámica de proyección de reservas a 5 años utilizando ciclos anidados y construyendo mapas estructurados.

### 3. Script Avanzado en AviatorScript

```javascript
// =========================================================================
// 1. ASSERTIONS Y VALIDACIONES DEFENSIVAS
// =========================================================================
if (polizaId == nil) { throw "El parámetro 'polizaId' es obligatorio"; }
if (seq.count(asegurados) == 0) { throw "La nómina de asegurados no puede estar vacía"; }
if (seq.count(historicoSiniestros) == 0) { throw "Se requiere historial de siniestros de al menos un período"; }

// =========================================================================
// 2. ANÁLISIS DE SINIESTRALIDAD HISTÓRICA CON PROGRAMACIÓN FUNCIONAL
// =========================================================================
let totalPrimasHist = seq.reduce(historicoSiniestros, lambda(acc, h) -> acc + h.montoPrimasCobradas end, 0.0);
let totalReclamacionesHist = seq.reduce(historicoSiniestros, lambda(acc, h) -> acc + h.montoReclamacionesPagadas end, 0.0);
let totalCasosGraves = seq.reduce(historicoSiniestros, lambda(acc, h) -> acc + h.casosGraves end, 0);

if (totalPrimasHist <= 0) { throw "Las primas históricas deben ser mayores a cero"; }
let lossRatioGlobal = totalReclamacionesHist / totalPrimasHist;

// Factor de ajuste por experiencia de siniestralidad (Experience Rating Factor)
let factorExperienciaSiniestral = 1.0;
if (lossRatioGlobal > 0.90) {
    factorExperienciaSiniestral = 1.35;
} elsif (lossRatioGlobal > 0.75) {
    factorExperienciaSiniestral = 1.15;
} elsif (lossRatioGlobal < 0.50) {
    factorExperienciaSiniestral = 0.88;
}

// =========================================================================
// 3. PROCESAMIENTO POBLACIONAL AVANZADO (LAMBDAS ANIDADAS & TRANSFORMACIONES)
// =========================================================================
let numAsegurados = seq.count(asegurados);
let sumaEdades = seq.reduce(asegurados, lambda(acc, a) -> acc + a.edad end, 0.0);
let promedioEdad = sumaEdades / numAsegurados;

// Cálculo de varianza de edad
let sumaDiferenciasCuadradas = seq.reduce(
    asegurados,
    lambda(acc, a) -> acc + math.pow(a.edad - promedioEdad, 2) end,
    0.0
);
let desviacionEstandarEdad = math.sqrt(sumaDiferenciasCuadradas / numAsegurados);

// Mapeo dinámico de riesgo individual con funciones anónimas complejas
let perfilesRiesgoIndividuales = seq.map(asegurados, lambda(a) ->
    let baseMortalidad = 0.001 * math.pow(1.06, a.edad - 18);
    let factorTabaquismo = a.fumador ? 1.80 : 1.0;
    
    // Conteo y ponderación de comorbilidades
    let numComorbilidades = seq.count(a.diagnosticosPrevios);
    let factorComorbilidad = 1.0 + (numComorbilidades * 0.25);
    
    // Evaluación de diagnósticos críticos específicos mediante seq.filter
    let tieneRiesgoCardiaco = seq.count(seq.filter(a.diagnosticosPrevios, lambda(d) -> d == "CARDIOPATIA" || d == "HIPERTENSION" end)) > 0;
    let recargoCritico = tieneRiesgoCardiaco ? 1.40 : 1.0;
    
    let primaRiesgoIndividual = 100000.0 * baseMortalidad * factorTabaquismo * factorComorbilidad * recargoCritico;

    return seq.map(
        "id", a.id,
        "edad", a.edad,
        "primaBaseCalculada", primaRiesgoIndividual,
        "nivelRiesgo", primaRiesgoIndividual > 5000.0 ? "ALTO" : (primaRiesgoIndividual > 2000.0 ? "MEDIO" : "BAJO")
    );
end);

// Sumatoria de primas puras individuales
let primaPuraTotalPoblacion = seq.reduce(perfilesRiesgoIndividuales, lambda(acc, p) -> acc + p.primaBaseCalculada end, 0.0);

// =========================================================================
// 4. APLICACIÓN DE FACTORES DE SECTOR, CARGOS Y MARGENES
// =========================================================================
let recargoSector = 1.0;
if (empresa.sectorRiesgo == "MINERIA_SUBTERRANEA") {
    recargoSector = 1.60;
} elsif (empresa.sectorRiesgo == "CONSTRUCCION_PESADA") {
    recargoSector = 1.35;
} elsif (empresa.sectorRiesgo == "TECNOLOGIA_OFICINA") {
    recargoSector = 0.90;
}

let primaTecnicaAjustada = primaPuraTotalPoblacion 
    * factorExperienciaSiniestral 
    * recargoSector 
    * parametrosEstrategicos.factorCargaCatastrofica;

// Aplicación de Recargos de Gastos y Margen (Cálculo Inverso Actuarial)
// PrimaComercial = PrimaTecnica / (1 - Gastos - Margen)
let tasaCargasTotales = parametrosEstrategicos.gastosAdministrativosTasa + parametrosEstrategicos.margenUtilidadEsperado;
if (tasaCargasTotales >= 1.0) { throw "La suma de gastos y margen no puede ser mayor o igual al 100%"; }

let primaComercialAnualTotal = primaTecnicaAjustada / (1.0 - tasaCargasTotales);
let primaMensualCapitaPromedio = (primaComercialAnualTotal / 12.0) / numAsegurados;

// =========================================================================
// 5. PROYECTO MATRICIAL DE RESERVAS TECNICAS A 5 AÑOS (CICLOS Y MATRICES)
// =========================================================================
let matrizProyeccionReservas = seq.list();
let acumuladoReservaMatematica = primaComercialAnualTotal * 0.20; // Reserva inicial (20%)

for anioProj in range(1, 6) {
    let factorInflacionMedica = math.pow(1.08, anioProj - 1);
    let primaProyectadaAnio = primaComercialAnualTotal * math.pow(1.03, anioProj - 1);
    let siniestrosEstimadosAnio = primaProyectadaAnio * lossRatioGlobal * factorInflacionMedica;
    let rendimientoInversionesReserva = acumuladoReservaMatematica * 0.05;
    
    acumuladoReservaMatematica = acumuladoReservaMatematica + (primaProyectadaAnio * 0.15) + rendimientoInversionesReserva - (siniestrosEstimadosAnio * 0.05);

    seq.add(matrizProyeccionReservas, seq.map(
        "anioProyeccion", anioProj,
        "primaProyectada", primaProyectadaAnio,
        "siniestralidadProyectada", siniestrosEstimadosAnio,
        "reservaMatematicaFinalPeriodo", acumuladoReservaMatematica
    ));
}

// Filtro de asegurados de alto riesgo para alertas de suscripción
let aseguradosAltoRiesgo = seq.filter(perfilesRiesgoIndividuales, lambda(p) -> p.nivelRiesgo == "ALTO" end);

// =========================================================================
// 6. ESTRUCTURA DE RETORNO COMPLEJA
// =========================================================================
return seq.map(
    "polizaId", polizaId,
    "resumenEstadisticoPoblacion", seq.map(
        "totalAsegurados", numAsegurados,
        "promedioEdad", promedioEdad,
        "desviacionEstandarEdad", desviacionEstandarEdad,
        "cantidadAltoRiesgo", seq.count(aseguradosAltoRiesgo)
    ),
    "analisisSiniestralidadHistorica", seq.map(
        "lossRatioGlobal", lossRatioGlobal,
        "factorExperienciaAplicado", factorExperienciaSiniestral,
        "totalCasosGravesHistoricos", totalCasosGraves
    ),
    "tarificacionFinanciera", seq.map(
        "primaPuraRiesgoTotal", primaPuraTotalPoblacion,
        "primaTecnicaAjustada", primaTecnicaAjustada,
        "primaComercialAnualTotal", primaComercialAnualTotal,
        "primaMensualPorCapitaPromedio", primaMensualCapitaPromedio
    ),
    "proyeccionReservasCincoAnios", matrizProyeccionReservas,
    "detalleAseguradosRiesgoAlto", aseguradosAltoRiesgo
);
```

## CASO 2: Motor de Tarificación Dinámica y Picos de Demanda para Redes Eléctricas Industriales (Utilities)

### 1. Descripción del Problema
Este motor procesa lecturas de telemetría por intervalos de tiempo (Smart Metering) aplicando:
1. **Agregación Funcional de Series Temporales**: Procesa listas de consumo por hora clasificados en tramos (Valle, Llano, Pico) con `seq.filter` y `seq.reduce`.
2. **Penalización por Exceso de Potencia Reactiva / Factor de Potencia ($\cos \phi$)**: Evalúa si el factor de potencia está por debajo de $0.92$ y aplica recargos progresivos no lineales.
3. **Cálculo de Demanda Máxima Coincidente (Peak Shaving)**: Detecta picos de potencia activa mediante ordenamiento dinámico de colecciones (`seq.sort`).
4. **Descuento por Inyección de Energía Renovable Autofacturada**: Liquida los exedentes fotovoltaicos inyectados a la red con tarifas de compensación indexadas.

### 2. Estructura de Parámetros de Entrada (JSON)

```json

  "contrato": {
    "potenciaContratadaKw": 500.0,
    "tipoTarifa": "GRAN_DEMANDA_3T",
    "exentoImpuestoCarbono": false
  },
  "lecturasIntervalo": [
    {"hora": 1, "bloque": "VALLE", "kwhConsumidos": 120.0, "kwPotenciaMaxima": 180.0, "kvarhReactiva": 30.0, "kwhInyectadosSolar": 0.0},
```

### 3. Script Avanzado en AviatorScript

```javascript
// =========================================================================
// 1. FILTRADO Y AGREGACIÓN DE ENERGÍA POR BLOQUES HORARIOS
// =========================================================================
let lecturasValle = seq.filter(lecturasIntervalo, lambda(l) -> l.bloque == "VALLE" end);
let lecturasLlano = seq.filter(lecturasIntervalo, lambda(l) -> l.bloque == "LLANO" end);
let lecturasPico  = seq.filter(lecturasIntervalo, lambda(l) -> l.bloque == "PICO" end);

let kwhTotalValle = seq.reduce(lecturasValle, lambda(acc, l) -> acc + l.kwhConsumidos end, 0.0);
let kwhTotalLlano = seq.reduce(lecturasLlano, lambda(acc, l) -> acc + l.kwhConsumidos end, 0.0);
let kwhTotalPico  = seq.reduce(lecturasPico,  lambda(acc, l) -> acc + l.kwhConsumidos end, 0.0);

let kwhConsumoBrutoTotal = kwhTotalValle + kwhTotalLlano + kwhTotalPico;

// Costos base de energía activa
let costoEnergiaValle = kwhTotalValle * preciosBase.kwhValle;
let costoEnergiaLlano = kwhTotalLlano * preciosBase.kwhLlano;
let costoEnergiaPico  = kwhTotalPico  * preciosBase.kwhPico;
let subtotalEnergiaActiva = costoEnergiaValle + costoEnergiaLlano + costoEnergiaPico;

// =========================================================================
// 2. ORDENAMIENTO DE DEMANDA MÁXIMA Y PENALIZACIÓN POR POTENCIA
// =========================================================================
// Ordenar lecturas por kwPotenciaMaxima descendente usando lambda en seq.sort
let lecturasOrdenadasPorPotencia = seq.sort(
    lecturasIntervalo,
    lambda(a, b) -> a.kwPotenciaMaxima > b.kwPotenciaMaxima ? -1 : (a.kwPotenciaMaxima < b.kwPotenciaMaxima ? 1 : 0) end
);

let lecturaPicoMaximo = lecturasOrdenadasPorPotencia[0];
let kwPotenciaMaximaRegistrada = lecturaPicoMaximo.kwPotenciaMaxima;

let costoTerminoPotenciaBase = contrato.potenciaContratadaKw * preciosBase.kwPotenciaMes;
let penalizacionExcesoPotencia = 0.0;

if (kwPotenciaMaximaRegistrada > contrato.potenciaContratadaKw) {
    let kwExceso = kwPotenciaMaximaRegistrada - contrato.potenciaContratadaKw;
    // Penalización no lineal (cuadrática sobre el exceso)
    penalizacionExcesoPotencia = (kwExceso * preciosBase.kwPotenciaMes * 2.0) + (math.pow(kwExceso, 1.2) * 1.5);
}

// =========================================================================
// 3. CÁLCULO DE FACTOR DE POTENCIA Y RECARGO REACTIVO GLOBAL
// =========================================================================
let totalKvarhReactiva = seq.reduce(lecturasIntervalo, lambda(acc, l) -> acc + l.kvarhReactiva end, 0.0);

// Factor de potencia aproximado = kWh / sqrt(kWh^2 + kVARh^2)
let denominadorS = math.sqrt(math.pow(kwhConsumoBrutoTotal, 2) + math.pow(totalKvarhReactiva, 2));
let factorDePotenciaGlobal = denominadorS == 0 ? 1.0 : (kwhConsumoBrutoTotal / denominadorS);

let penalizacionReactiva = 0.0;
if (factorDePotenciaGlobal < 0.92) {
    let diferenciaFP = 0.92 - factorDePotenciaGlobal;
    penalizacionReactiva = subtotalEnergiaActiva * (diferenciaFP * 2.5); // 2.5% de recargo por cada punto porcentual
}

// =========================================================================
// 4. COMPENSACIÓN POR AUTOCONSUMO RENOVABLE
// =========================================================================
let totalKwhInyectadosSolar = seq.reduce(lecturasIntervalo, lambda(acc, l) -> acc + l.kwhInyectadosSolar end, 0.0);
let valorCompensacionSolar = totalKwhInyectadosSolar * preciosBase.compensacionSolarKwh;

// =========================================================================
// 5. CONSOLIDACIÓN DE IMPUESTOS Y TOTAL FACTURA
// =========================================================================
let subtotalFacturaBruto = subtotalEnergiaActiva + costoTerminoPotenciaBase + penalizacionExcesoPotencia + penalizacionReactiva;
let subtotalDespuesCompensacion = math.max(0.0, subtotalFacturaBruto - valorCompensacionSolar);

let impuestoCarbono = contrato.exentoImpuestoCarbono ? 0.0 : (kwhConsumoBrutoTotal * 0.0035);
let impuestoElectricoRegional = subtotalDespuesCompensacion * 0.0510;
let iva = (subtotalDespuesCompensacion + impuestoCarbono + impuestoElectricoRegional) * 0.12;

let totalPagarFactura = subtotalDespuesCompensacion + impuestoCarbono + impuestoElectricoRegional + iva;

return seq.map(
    "puntoSuministroId", puntoSuministroId,
    "periodo", periodoFacturacion,
    "metricasConsumo", seq.map(
        "totalKwhConsumidos", kwhConsumoBrutoTotal,
        "potenciaMaximaRegistradaKw", kwPotenciaMaximaRegistrada,
        "factorDePotenciaCalculado", factorDePotenciaGlobal,
        "totalKwhInyectadosSolar", totalKwhInyectadosSolar
    ),
    "desgloseCargos", seq.map(
        "energiaActivaValle", costoEnergiaValle,
        "energiaActivaLlano", costoEnergiaLlano,
        "energiaActivaPico", costoEnergiaPico,
        "terminoPotenciaBase", costoTerminoPotenciaBase,
        "penalizacionExcesoPotencia", penalizacionExcesoPotencia,
        "penalizacionBajoFactorPotencia", penalizacionReactiva,
        "creditoAutoconsumoSolar", valorCompensacionSolar
    ),
    "impuestosyTasas", seq.map(
        "impuestoCarbono", impuestoCarbono,
        "impuestoElectrico", impuestoElectricoRegional,
        "iva", iva
    ),
    "totalFinalAPagar", totalPagarFactura
);
```

```sql
    'Tarifación Dinámica de Energía con Penalización FP y Solar',
    1,
    $$
    let kwhValle = seq.reduce(seq.filter(lecturasIntervalo, lambda(l) -> l.bloque == "VALLE" end), lambda(acc, l) -> acc + l.kwhConsumidos end, 0.0);
    let kwhLlano = seq.reduce(seq.filter(lecturasIntervalo, lambda(l) -> l.bloque == "LLANO" end), lambda(acc, l) -> acc + l.kwhConsumidos end, 0.0);
    let kwhPico  = seq.reduce(seq.filter(lecturasIntervalo, lambda(l) -> l.bloque == "PICO" end),  lambda(acc, l) -> acc + l.kwhConsumidos end, 0.0);

    let costoEnergia = (kwhValle * preciosBase.kwhValle) + (kwhLlano * preciosBase.kwhLlano) + (kwhPico * preciosBase.kwhPico);
    let totalKwh = kwhValle + kwhLlano + kwhPico;
    let totalReactiva = seq.reduce(lecturasIntervalo, lambda(acc, l) -> acc + l.kvarhReactiva end, 0.0);

    let fp = totalKwh / math.sqrt(math.pow(totalKwh, 2) + math.pow(totalReactiva, 2));
    let penalizacionFP = fp < 0.92 ? costoEnergia * ((0.92 - fp) * 2.5) : 0.0;

    let subtotal = costoEnergia + (contrato.potenciaContratadaKw * preciosBase.kwPotenciaMes) + penalizacionFP;
    let totalConIva = subtotal * 1.12;

    return seq.map(
        "puntoSuministroId", puntoSuministroId,
        "totalKwh", totalKwh,
        "factorPotencia", fp,
        "costoEnergia", costoEnergia,
        "penalizacionFP", penalizacionFP,
        "totalPagar", totalConIva
    );
    $$,
    TRUE,
    CURRENT_TIMESTAMP
);
```

## CASO 3: Motor de Distribución de Riesgo de Reaseguro por Tratado Excess of Loss (XoL) Layered

### 1. Descripción del Problema
Este motor distribuye siniestros catastróficos a través de una torre de capas de reaseguro (Excess of Loss Layers):
1. **Modelado Dinámico de Capas (Layers)**: Filtra y procesa la retención de la aseguradora cedente y Múltiples tramos (Tranches) de reaseguradoras.
2. **Cálculo de Restablecimientos de Capas (Reinstatements)**: Calcula el costo de reinstalación de límite disponible si una capa es erosionada por un siniestro previo en la misma vigencia.
3. **Distribución Proporcional de Gastos de Ajuste (LAE - Loss Adjustment Expenses)**: Asigna gastos legales y peritaje proporcionalmente a la pérdida absorbida por cada capa (Cláusula Pro-Rata LAE).

### 3. Script Avanzado en AviatorScript

```javascript
// =========================================================================
// 1. VALIDACIÓN Y PRECONDICIONES ACTUARIALES
// =========================================================================
if (montoSiniestroBruto <= 0) { throw "El monto del siniestro debe ser mayor que cero"; }
if (retencionCedente < 0) { throw "La retención no puede ser negativa"; }

let saldoPerdidaRemanente = montoSiniestroBruto;
let montoAbsorbidoCedente = math.min(saldoPerdidaRemanente, retencionCedente);
saldoPerdidaRemanente = saldoPerdidaRemanente - montoAbsorbidoCedente;

let repartoCapasResultado = seq.list();
let totalAbsorbidoReaseguradores = 0.0;
let totalPrimasReinstalamiento = 0.0;

// =========================================================================
// 2. PROCESAMIENTO ITERATIVO DE LA TORRE DE CAPAS (LAYERS)
// =========================================================================
for capa in capasReaseguro {
    let lossAbsorbidaCapa = 0.0;
    let costoReinstalamientoCapa = 0.0;

    if (saldoPerdidaRemanente > 0) {
        // Verificar si la pérdida alcanza la prioridad/deducible de esta capa
        if (montoSiniestroBruto > capa.prioridadExceso) {
            let excesoSobrePrioridad = montoSiniestroBruto - capa.prioridadExceso;
            lossAbsorbidaCapa = math.min(excesoSobrePrioridad, capa.limiteCapa);
            
            // Deducción de lo ya procesado
            saldoPerdidaRemanente = saldoPerdidaRemanente - lossAbsorbidaCapa;
            totalAbsorbidoReaseguradores = totalAbsorbidoReaseguradores + lossAbsorbidaCapa;

            // Cálculo de prima de reinstalamiento (Reinstatement Premium)
            let proporcionCapaConsumida = lossAbsorbidaCapa / capa.limiteCapa;
            costoReinstalamientoCapa = proporcionCapaConsumida * capa.primaBaseCapa * capa.porcentajeReinstalamiento;
            totalPrimasReinstalamiento = totalPrimasReinstalamiento + costoReinstalamientoCapa;
        }
    }

    // Asignación Pro-Rata de Gastos de Ajuste (LAE)
    let proporcionLossSobreBruto = lossAbsorbidaCapa / montoSiniestroBruto;
    let laeAsignadoCapa = gastosAjusteLAE * proporcionLossSobreBruto;

    seq.add(repartoCapasResultado, seq.map(
        "capaNumero", capa.capaNumero,
        "coberturaCapa", string.builder().append(capa.limiteCapa).append(" xs ").append(capa.prioridadExceso).toString(),
        "perdidaAbsorbida", lossAbsorbidaCapa,
        "gastosLAEAsignados", laeAsignadoCapa,
        "primaReinstalamientoACobrar", costoReinstalamientoCapa,
        "capaAgotada", lossAbsorbidaCapa == capa.limiteCapa
    ));
}

// =========================================================================
// 3. RETENCIÓN FINAL RETENIDA Y RESUMEN ESTRUCTURADO
// =========================================================================
let laeAsignadoCedente = gastosAjusteLAE * (montoAbsorbidoCedente / montoSiniestroBruto);
let costoTotalEfectivoCedente = montoAbsorbidoCedente + laeAsignadoCedente + totalPrimasReinstalamiento + saldoPerdidaRemanente;

return seq.map(
    "siniestroId", siniestroId,
    "resumenSiniestro", seq.map(
        "montoSiniestroBruto", montoSiniestroBruto,
        "gastosAjusteLAETotal", gastosAjusteLAE,
        "excesoNoCubiertoTorre", saldoPerdidaRemanente
    ),
    "posicionCedenteAseguradora", seq.map(
        "retencionBaseAbsorbida", montoAbsorbidoCedente,
        "laeCedente", laeAsignadoCedente,
        "totalPrimasReinstalamientoPAGADAS", totalPrimasReinstalamiento,
        "costoTotalNetoParaCedente", costoTotalEfectivoCedente
    ),
    "detalleCapasReaseguro", repartoCapasResultado
);
```
